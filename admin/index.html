// 1. جلب البطاقات من gift_card_designs وعرضها مع السعر async function loadDynamicGiftCards(supabaseClient, containerEl, onSelectCallback) { const { data, error } = await supabaseClient.from('gift_card_designs').select('*'); if (error || !data || !data.length) return; containerEl.innerHTML = data.map((card, index) => { const title = card.title || card.name || card.label || `Card ${index + 1}`; const imgUrl = card.image_url || card.url || ''; const price = Number(card.price ?? card.extra_price ?? 0); const priceText = price > 0 ? `+${price.toFixed(2)} JOD` : 'Free'; return ` `; }).join(''); containerEl.querySelectorAll('.gift-card-option').forEach(btn => { btn.addEventListener('click', () => { containerEl.querySelectorAll('.gift-card-option').forEach(b => b.classList.remove('border-[#00313C]', 'ring-2', 'ring-[#00313C]/20') ); btn.classList.add('border-[#00313C]', 'ring-2', 'ring-[#00313C]/20'); if (typeof onSelectCallback === 'function') { onSelectCallback({ id: btn.dataset.cardId, title: btn.dataset.cardTitle, image_url: btn.dataset.cardImage, price: parseFloat(btn.dataset.cardPrice) || 0 }); } }); }); } // 2. حفظ البطاقة المختارة ضمن بيانات السلة وحساب سعرها في الإجمالي function applyGiftCardToCartState(cartState, selectedCard, messageText) { const cardFee = selectedCard ? (parseFloat(selectedCard.price) || 0) : 0; cartState.giftCard = selectedCard ? { id: selectedCard.id, title: selectedCard.title, image_url: selectedCard.image_url, price: cardFee, message: (messageText || '').trim() } : null; const itemsSubtotal = (cartState.items || []).reduce((sum, item) => { const unit = parseFloat(item.price) || 0; const qty = parseInt(item.qty || item.quantity || 1, 10); return sum + (unit * qty); }, 0); cartState.giftCardTotal = cardFee; cartState.grandTotal = itemsSubtotal + cardFee; return cartState; }
